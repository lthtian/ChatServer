# ABOUTME: Builds the image processing library independently of the chat server.
# ABOUTME: Registers real image and filesystem tests for the media components.
cmake_minimum_required(VERSION 3.16)
project(chat_media LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)

add_library(chat_media src/image.cpp src/disk_store.cpp)
target_include_directories(chat_media PUBLIC include)
target_include_directories(chat_media SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(chat_media PUBLIC ${OpenCV_LIBS})
if(MSVC)
    target_compile_options(chat_media PRIVATE /W4 /utf-8)
else()
    target_compile_options(chat_media PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(chat_image src/image_tool.cpp)
target_include_directories(chat_image PRIVATE ../thirdparty)
target_link_libraries(chat_image PRIVATE chat_media)
if(MSVC)
    target_compile_options(chat_image PRIVATE /W4 /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(image_test tests/image_test.cpp)
    target_include_directories(image_test SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(image_test PRIVATE chat_media)
    if(MSVC)
        target_compile_options(image_test PRIVATE /W4 /utf-8)
    endif()
    foreach(scenario jpeg alpha small orientation corrupt byte_limit pixel_limit format)
        add_test(NAME image_${scenario} COMMAND image_test ${scenario})
    endforeach()
    add_executable(disk_store_test tests/disk_store_test.cpp)
    target_link_libraries(disk_store_test PRIVATE chat_media)
    if(MSVC)
        target_compile_options(disk_store_test PRIVATE /W4 /utf-8)
    endif()
    foreach(scenario publish abandon limit immutable key)
        add_test(NAME disk_${scenario} COMMAND disk_store_test ${scenario})
    endforeach()
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME image_tool COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/image_tool_test.py $<TARGET_FILE:chat_image>)
endif()
