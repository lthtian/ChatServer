cmake_minimum_required(VERSION 3.20)
project(media_process LANGUAGES CXX)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core)

set(FFMPEG_ROOT "" CACHE PATH "FFmpeg development package containing include, lib and bin")
if(WIN32)
  find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h
    PATHS "${FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)
endif()
add_executable(media_process main.cpp media_io.cpp remux.cpp transcoder.cpp hls_package.cpp)
target_compile_features(media_process PRIVATE cxx_std_17)
set_target_properties(media_process PROPERTIES CXX_EXTENSIONS OFF)
if(WIN32)
target_include_directories(media_process SYSTEM PRIVATE "${FFMPEG_INCLUDE_DIR}")
endif()
target_link_libraries(media_process PRIVATE Qt5::Core)
if(WIN32)
foreach(component avformat avcodec avutil swscale swresample)
  find_library(FFMPEG_${component}_LIBRARY NAMES ${component}
    PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(media_process PRIVATE "${FFMPEG_${component}_LIBRARY}")
endforeach()
else()
  target_link_libraries(media_process PRIVATE PkgConfig::FFMPEG)
endif()
if(MSVC)
  target_compile_options(media_process PRIVATE /W4 /utf-8)
else()
  target_compile_options(media_process PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_compile_definitions(media_process PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
  file(GLOB ffmpeg_runtime "${FFMPEG_ROOT}/bin/*.dll")
  add_custom_command(TARGET media_process POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${ffmpeg_runtime} "$<TARGET_FILE:Qt5::Core>"
      "$<TARGET_FILE_DIR:media_process>" COMMAND_EXPAND_LISTS VERBATIM)
endif()
if(MINGW)
  target_link_options(media_process PRIVATE -municode)
  get_filename_component(compiler_directory "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(runtime libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
    add_custom_command(TARGET media_process POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${compiler_directory}/${runtime}"
        "$<TARGET_FILE_DIR:media_process>" VERBATIM)
  endforeach()
endif()
